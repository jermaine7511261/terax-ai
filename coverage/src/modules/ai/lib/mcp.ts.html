import { invoke } from "@tauri-apps/api/core";
import { LazyStore } from "@tauri-apps/plugin-store";
import { z } from "zod";
 
/**
 * MCP client frontend: server config persistence, Tauri command wrappers, and
 * the JSON-Schema → zod converter used to build dynamic tools from remote
 * `tools/list` schemas.
 *
 * All MCP tools are treated as untrusted: `buildMcpTools` marks them
 * `needsApproval: true` and the backend never auto-executes anything — every
 * call is a JSON-RPC forward to the configured server.
 */
 
export type McpTransport = "stdio" | "http";
 
export type McpServerConfig = {
  id: string;
  name: string;
  transport: McpTransport;
  command: string;
  args: string[];
  cwd: string;
  env: Record<string, string>;
  url: string;
  headers: Record<string, string>;
};
 
export type McpToolInfo = {
  server_id: string;
  server_name: string;
  name: string;
  description: string;
  input_schema: unknown;
};
 
export type McpServerStatus = {
  id: string;
  name: string;
  connected: boolean;
  tool_count: number;
  stderr_tail: string;
  error: string | null;
};
 
const STORE_PATH = "yamet-ai-mcp.json";
const KEY_SERVERS = "servers";
 
const store = new LazyStore(STORE_PATH, { defaults: {}, autoSave: 200 });
 
export async function loadMcpServers(): Promise<McpServerConfig[]> {
  return (await store.get<McpServerConfig[]>(KEY_SERVERS)) ?? [];
}
 
export async function saveMcpServers(list: McpServerConfig[]): Promise<void> {
  await store.set(KEY_SERVERS, list);
  await store.save();
}
 
export function newMcpServerId(): string {
  return `mcp-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 6)}`;
}
 
/** Normalize an arbitrary string to `[a-z0-9_]+` (tool-name component). */
function slug(s: string): string {
  return s
    .toLowerCase()
    .replace(/[^a-z0-9_]+/g, "_")
    .replace(/^_+|_+$/g, "");
}
 
/**
 * MCP tool name convention: `mcp_<serverId>_<toolName>`, sanitized to
 * `[a-zA-Z0-9_]+` so it is model-visible, unique across servers, and
 * parseable by the tool card (which branches on the `mcp_` prefix).
 */
export function sanitizeMcpToolName(serverId: string, toolName: string): string {
  const id = slug(serverId) || "server";
  const t = slug(toolName) || "tool";
  return `mcp_${id}_${t}`;
}
 
/**
 * Convert an MCP tool inputSchema (JSON Schema subset) to a zod schema.
 * Unsupported / missing shapes fall back to a permissive object so a remote
 * tool never breaks the local tool registry. Unknown keys are allowed
 * (`.passthrough()`) — remote servers may add fields the schema omits.
 */
export function jsonSchemaToZod(schema: unknown): z.ZodTypeAny {
  if (!schema || typeof schema !== "object")
    return z.record(z.string(), z.unknown());
  const s = schema as Record<string, unknown>;
  const description =
    typeof s.description === "string" ? (s.description as string) : undefined;
 
  switch (s.type) {
    case "string": {
      if (Array.isArray(s.enum) && s.enum.length > 0) {
        return z.enum(s.enum as [string, ...string[]]);
      }
      const zs = description ? z.string().describe(description) : z.string();
      return zs;
    }
    case "number": {
      return description ? z.number().describe(description) : z.number();
    }
    case "integer": {
      return description ? z.number().int().describe(description) : z.number().int();
    }
    case "boolean": {
      return description ? z.boolean().describe(description) : z.boolean();
    }
    case "array": {
      const items = s.items ? jsonSchemaToZod(s.items) : z.unknown();
      return description ? z.array(items).describe(description) : z.array(items);
    }
    case "object": {
      const props = (s.properties ?? {}) as Record<string, unknown>;
      const required = new Set<string>(
        Array.isArray(s.required)
          ? (s.required as unknown[]).filter(
              (r): r is string => typeof r === "string",
            )
          : [],
      );
      const shape: Record<string, z.ZodTypeAny> = {};
      for (const [k, v] of Object.entries(props)) {
        const field = jsonSchemaToZod(v);
        shape[k] = required.has(k) ? field : field.optional();
      }
      const obj = z.object(shape).passthrough();
      return description ? obj.describe(description) : obj;
    }
    default: {
      // Null / oneOf / anyOf / untyped: permissive object.
      return description
        ? z.record(z.string(), z.unknown()).describe(description)
        : z.record(z.string(), z.unknown());
    }
  }
}
